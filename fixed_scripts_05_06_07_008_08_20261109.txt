# =============================================================================
# 05_foreign_price_process.R
#
# Parameter estimated:
#   rhoF
#
# Model equation:
#   pif_t = rhoF * pif_{t-1} + eps_pf,t
#
# Main empirical design:
#   1. Estimate quarterly persistence from the U.S. BEA/FRED export-price index
#      for industrial supplies and materials (B639RG3Q086SBEA).
#   2. Validate that quarterly proxy against Brazil-specific annual import-price
#      indexes from Serra (1982), Table 13.
#   3. Use Serra's intermediate-goods import-price index as the PRIMARY
#      historical validation target because it is the closest empirical match to
#      the model's imported-intermediate-input price concept.
#   4. Keep Serra's general import-price index as a SECONDARY validation target.
#   5. Estimate the Brazilian annual AR(1) only as a diagnostic. It is NOT used
#      as the quarterly Dynare rhoF because temporal aggregation changes the
#      stochastic process.
#
# Baseline quarterly proxy:
#   B639RG3Q086SBEA = U.S. exports of industrial supplies and materials,
#                     chain-type price index.
#
# Robustness / diagnostics:
#   B686RG3Q086SBEA = durable industrial supplies and materials
#   B687RG3Q086SBEA = nondurable industrial supplies and materials
#   B640RG3Q086SBEA = capital goods except automotive
#   USA_Exports.csv = optional broad U.S. export-price benchmark
#
# Brazilian annual validation database:
#   data/processed/brazil_import_price_serra_1963_1981.csv
#
# IMPORTANT:
#   - Quarterly rhoF comes from quarterly B639 inflation.
#   - Serra annual data validate the economic mapping.
#   - Do NOT convert Serra's annual rho into a quarterly rho by fourth-rooting.
# =============================================================================

# Read a quarterly FRED/BEA price index
# -----------------------------------------------------------------------------

read_fred_price <- function(path, short_name) {
  
  x <- read.csv(
    path,
    check.names = FALSE,
    na.strings = c("", "NA", ".")
  )
  
  if (nrow(x) == 0) {
    stop(
      "File contains no observations: ",
      path
    )
  }
  
  date_candidates <- names(x)[
    tolower(names(x)) %in%
      c("observation_date", "date")
  ]
  
  if (length(date_candidates) == 0) {
    stop(
      "Could not identify the date column in ",
      path,
      ". Expected observation_date or DATE."
    )
  }
  
  date_col <- date_candidates[1]
  
  dates <- as.Date(x[[date_col]])
  
  if (any(is.na(dates))) {
    stop(
      "Could not parse all dates in: ",
      path
    )
  }
  
  candidate_cols <- setdiff(
    names(x),
    date_col
  )
  
  if (length(candidate_cols) == 0) {
    stop(
      "No price-index column found in: ",
      path
    )
  }
  
  numeric_counts <- vapply(
    candidate_cols,
    function(nm) {
      
      sum(
        !is.na(
          suppressWarnings(
            as.numeric(x[[nm]])
          )
        )
      )
    },
    numeric(1)
  )
  
  if (max(numeric_counts) == 0) {
    stop(
      "No numeric price series found in: ",
      path
    )
  }
  
  value_col <- candidate_cols[
    which.max(numeric_counts)
  ]
  
  values <- suppressWarnings(
    as.numeric(x[[value_col]])
  )
  
  year <- as.integer(
    format(dates, "%Y")
  )
  
  month <- as.integer(
    format(dates, "%m")
  )
  
  qtr <- ((month - 1L) %/% 3L) + 1L
  
  out <- data.frame(
    date = dates,
    year = year,
    qtr = qtr,
    quarter = quarter_string(
      year,
      qtr
    ),
    stringsAsFactors = FALSE
  )
  
  out[[short_name]] <- values
  
  out <- out[
    order(out$date),
  ]
  
  inflation_name <- paste0(
    short_name,
    "_log_inflation"
  )
  
  out[[inflation_name]] <- c(
    NA_real_,
    diff(
      log(out[[short_name]])
    )
  )
  
  out
}


# -----------------------------------------------------------------------------
# Annualize quarterly price index
#
# avg = annual average of four quarterly index observations
# q4  = Q4 price index
# -----------------------------------------------------------------------------

annualize_price <- function(
    qdata,
    price_col,
    proxy_name,
    method = c("avg", "q4")
) {
  
  method <- match.arg(method)
  
  years <- sort(
    unique(qdata$year)
  )
  
  rows <- vector(
    "list",
    length(years)
  )
  
  for (i in seq_along(years)) {
    
    y <- years[i]
    
    z <- qdata[
      qdata$year == y,
      c(
        "year",
        "qtr",
        price_col
      ),
      drop = FALSE
    ]
    
    names(z)[3] <- "price"
    
    complete_four <-
      sum(!is.na(z$price)) == 4
    
    if (method == "avg") {
      
      annual_price <-
        if (complete_four) {
          mean(
            z$price,
            na.rm = TRUE
          )
        } else {
          NA_real_
        }
      
    } else {
      
      q4_value <-
        z$price[z$qtr == 4]
      
      annual_price <-
        if (
          complete_four &&
          length(q4_value) == 1
        ) {
          q4_value
        } else {
          NA_real_
        }
    }
    
    rows[[i]] <- data.frame(
      proxy = proxy_name,
      annualization = method,
      year = y,
      us_price_index = annual_price,
      stringsAsFactors = FALSE
    )
  }
  
  out <- do.call(
    rbind,
    rows
  )
  
  out <- out[
    order(out$year),
  ]
  
  out$us_log_inflation <- c(
    NA_real_,
    diff(
      log(out$us_price_index)
    )
  )
  
  row.names(out) <- NULL
  
  out
}


# -----------------------------------------------------------------------------
# Safe correlation
# -----------------------------------------------------------------------------

safe_cor <- function(
    x,
    y,
    method
) {
  
  ok <- complete.cases(
    x,
    y
  )
  
  x <- x[ok]
  y <- y[ok]
  
  if (
    length(x) < 3 ||
    sd(x) == 0 ||
    sd(y) == 0
  ) {
    return(
      NA_real_
    )
  }
  
  cor(
    x,
    y,
    method = method
  )
}

# -----------------------------------------------------------------------------
# Pearson and Spearman validation
# -----------------------------------------------------------------------------

validation_correlations <- function(
    validation_data,
    proxy_name,
    proxy_role,
    annualization_method,
    target_name,
    target_role,
    target_column,
    periods
) {
  
  z <- validation_data[
    validation_data$proxy == proxy_name &
      validation_data$annualization ==
      annualization_method,
    ,
    drop = FALSE
  ]
  
  result <- vector(
    "list",
    nrow(periods)
  )
  
  for (i in seq_len(nrow(periods))) {
    
    start_year <-
      periods$start_year[i]
    
    end_year <-
      periods$end_year[i]
    
    s <- z[
      z$year >= start_year &
        z$year <= end_year,
      ,
      drop = FALSE
    ]
    
    ok <- complete.cases(
      s$us_log_inflation,
      s[[target_column]]
    )
    
    result[[i]] <- data.frame(
      
      proxy = proxy_name,
      
      proxy_role = proxy_role,
      
      brazil_target = target_name,
      
      target_role = target_role,
      
      annualization =
        annualization_method,
      
      period = period_label(
        start_year,
        end_year
      ),
      
      n = sum(ok),
      
      pearson_correlation =
        safe_cor(
          s$us_log_inflation,
          s[[target_column]],
          "pearson"
        ),
      
      spearman_correlation =
        safe_cor(
          s$us_log_inflation,
          s[[target_column]],
          "spearman"
        ),
      
      stringsAsFactors = FALSE
    )
  }
  
  do.call(
    rbind,
    result
  )
}


# -----------------------------------------------------------------------------
# Quarterly AR(1)
# -----------------------------------------------------------------------------

fit_ar1 <- function(
    data,
    variable,
    start_year,
    start_qtr,
    end_year,
    end_qtr,
    proxy,
    role,
    recommended = "NO"
) {
  
  z <- data[
    order(
      data$year,
      data$qtr
    ),
    c(
      "year",
      "qtr",
      variable
    ),
    drop = FALSE
  ]
  
  names(z)[3] <- "x"
  
  # IMPORTANT:
  # create lag before selecting sample.
  z$lag_x <- c(
    NA_real_,
    head(
      z$x,
      -1
    )
  )
  
  start_index <-
    start_year * 4 +
    start_qtr
  
  end_index <-
    end_year * 4 +
    end_qtr
  
  z_index <-
    z$year * 4 +
    z$qtr
  
  z <- z[
    z_index >= start_index &
      z_index <= end_index &
      complete.cases(
        z$x,
        z$lag_x
      ),
    ,
    drop = FALSE
  ]
  
  if (nrow(z) < 8) {
    stop(
      "Too few observations for AR(1): ",
      proxy,
      " ",
      role
    )
  }
  
  model <- lm(
    x ~ lag_x,
    data = z
  )
  
  coefs <- coef(model)
  
  vc_ols <- vcov(model)
  
  vc_hc1 <- sandwich::vcovHC(
    model,
    type = "HC1"
  )
  
  vc_nw <- sandwich::NeweyWest(
    model,
    lag = 4,
    prewhite = FALSE,
    adjust = TRUE
  )
  
  intercept <- unname(
    coefs["(Intercept)"]
  )
  
  rho <- unname(
    coefs["lag_x"]
  )
  
  implied_mean <-
    if (
      is.finite(rho) &&
      abs(1 - rho) > 1e-10
    ) {
      
      intercept /
        (1 - rho)
      
    } else {
      
      NA_real_
    }
  
  half_life <-
    if (
      is.finite(rho) &&
      rho > 0 &&
      rho < 1
    ) {
      
      log(0.5) /
        log(rho)
      
    } else {
      
      NA_real_
    }
  
  lb4 <- Box.test(
    residuals(model),
    lag = 4,
    type = "Ljung-Box",
    fitdf = 1
  )
  
  lb8 <- Box.test(
    residuals(model),
    lag = 8,
    type = "Ljung-Box",
    fitdf = 1
  )
  
  summary_row <- data.frame(
    
    proxy = proxy,
    
    role = role,
    
    sample = paste0(
      start_year,
      "Q",
      start_qtr,
      "-",
      end_year,
      "Q",
      end_qtr
    ),
    
    n = nrow(z),
    
    intercept = intercept,
    
    rhoF = rho,
    
    standard_error =
      sqrt(
        vc_ols[
          "lag_x",
          "lag_x"
        ]
      ),
    
    robust_se_HC1 =
      sqrt(
        vc_hc1[
          "lag_x",
          "lag_x"
        ]
      ),
    
    newey_west_se =
      sqrt(
        vc_nw[
          "lag_x",
          "lag_x"
        ]
      ),
    
    shock_sd =
      sigma(model),
    
    implied_unconditional_mean =
      implied_mean,
    
    half_life_quarters =
      half_life,
    
    ljung_box_p_lag4 =
      unname(
        lb4$p.value
      ),
    
    ljung_box_p_lag8 =
      unname(
        lb8$p.value
      ),
    
    recommended_baseline =
      recommended,
    
    stringsAsFactors = FALSE
  )
  
  list(
    summary = summary_row,
    model = model,
    data = z
  )
}

# -----------------------------------------------------------------------------
# Residual bootstrap
# -----------------------------------------------------------------------------

bootstrap_ar1_rho <- function(
    model,
    observed_x,
    B = 5000,
    seed = 1964
) {
  
  set.seed(seed)
  
  a_hat <- unname(
    coef(model)[
      "(Intercept)"
    ]
  )
  
  rho_hat <- unname(
    coef(model)[
      "lag_x"
    ]
  )
  
  residual_pool <-
    residuals(model) -
    mean(
      residuals(model)
    )
  
  n <- length(
    observed_x
  )
  
  draws <- rep(
    NA_real_,
    B
  )
  
  for (b in seq_len(B)) {
    
    x_star <- rep(
      NA_real_,
      n
    )
    
    x_star[1] <-
      observed_x[1]
    
    for (t in 2:n) {
      
      u_star <- sample(
        residual_pool,
        size = 1,
        replace = TRUE
      )
      
      x_star[t] <-
        a_hat +
        rho_hat *
        x_star[t - 1] +
        u_star
    }
    
    boot_df <- data.frame(
      
      x =
        x_star[
          2:n
        ],
      
      lag_x =
        x_star[
          1:(n - 1)
        ]
    )
    
    boot_model <- lm(
      x ~ lag_x,
      data = boot_df
    )
    
    draws[b] <- unname(
      coef(boot_model)[
        "lag_x"
      ]
    )
  }
  
  draws
}

upsert_dictionary_row <- function(
    df,
    key_col,
    key_value,
    values
) {
  
  if (
    !(key_col %in%
      names(df))
  ) {
    return(df)
  }
  
  
  idx <- which(
    as.character(
      df[[key_col]]
    ) ==
      key_value
  )
  
  
  if (
    length(idx) == 0
  ) {
    
    new_row <-
      as.list(
        rep(
          NA_character_,
          ncol(df)
        )
      )
    
    names(new_row) <-
      names(df)
    
    new_row[[key_col]] <-
      key_value
    
    
    for (
      nm in
      names(values)
    ) {
      
      if (
        nm %in%
        names(df)
      ) {
        
        new_row[[nm]] <-
          values[[nm]]
      }
    }
    
    
    df <- rbind(
      df,
      as.data.frame(
        new_row,
        stringsAsFactors = FALSE
      )
    )
    
  } else {
    
    idx <- idx[1]
    
    
    for (
      nm in
      names(values)
    ) {
      
      if (
        nm %in%
        names(df)
      ) {
        
        df[
          idx,
          nm
        ] <- values[[nm]]
      }
    }

# =============================================================================
# 06_trade_steady_state_shares.R
# =============================================================================
# Purpose
# -------
# Calibrate the pre-oil-shock trade shares used in the DSGE:
#
#   x_y  = exports / GDP
#   cf_y = imported final-consumption goods / GDP
#   z_y  = imported intermediate goods / GDP
#
# and, when g_y and ig_y are already available:
#
#   cH_y = 1 - x_y - g_y - ig_y
#
# Baseline calibration window: 1968-1972
# Robustness window:          1967-1972
#
# IMPORTANT DESIGN CHANGES IN THIS VERSION
# ----------------------------------------
# 1. The IBGE national-accounts workbook is read from the exact historical sheet
#    I.2.1 and the exact published labels:
#       "Produto Interno Bruto"
#       "Exportacao de bens e servicos"
#       "Menos: Importacao de bens e servicos"
#
# 2. The script DOES NOT try to guess generic series names such as
#    exports_fob_usd_million in the external-sector workbook.
#
# 3. The external-sector FOB workbook is now OPTIONAL. It is used only as an
#    additional historical diagnostic. If it cannot be parsed, the calibration
#    still runs because x_y, M/Y, cf_y and z_y do not depend on it.
#
# 4. The parser detects whether years run across a row or down a column and is
#    written for old mixed-format Excel files.
#
# Economic mapping
# ----------------
# The national accounts give same-unit nominal ratios:
#
#   x_y,t = X_t / Y_t
#   m_y,t = M_t / Y_t
#
# Doellinger et al. (1974), Table VI.2 gives merchandise-import composition:
#
#   sC_t = consumer imports / merchandise imports
#   sZ_t = intermediate imports / merchandise imports
#
# We therefore construct the calibration bridge:
#
#   cf_y,t = m_y,t * sC_t
#   z_y,t  = m_y,t * sZ_t
#
# Capital-goods imports remain separate diagnostics. They are NOT folded into
# z_y because the current model's Z block represents imported intermediate
# inputs rather than private imported capital.
# =============================================================================

first_existing <- function(paths, label, required = TRUE) {
  hit <- paths[file.exists(paths)]
  
  if (length(hit) > 0) {
    return(hit[1])
  }
  
  if (required) {
    stop(
      "Could not find ", label, ". Tried:\n - ",
      paste(paths, collapse = "\n - ")
    )
  }
  
  NA_character_
}

normalize_text <- function(x) {
  x <- as.character(x)
  x[is.na(x)] <- ""
  
  y <- suppressWarnings(
    iconv(x, from = "", to = "ASCII//TRANSLIT")
  )
  
  y[is.na(y)] <- x[is.na(y)]
  y <- tolower(y)
  y <- gsub("[^a-z0-9]+", " ", y)
  y <- trimws(gsub("[[:space:]]+", " ", y))
  y
}

as_number_safe <- function(x) {
  # readxl usually returns numeric Excel cells as numeric already.
  if (is.numeric(x)) {
    return(as.numeric(x))
  }
  
  s <- trimws(as.character(x))
  s[is.na(x)] <- NA_character_
  s[s %in% c("", "NA", "N/A", "-", "--")] <- NA_character_
  
  out <- suppressWarnings(as.numeric(s))
  missing <- which(is.na(out) & !is.na(s))
  
  if (length(missing) > 0) {
    for (i in missing) {
      z <- s[i]
      
      # Remove common currency/text clutter without assuming a particular unit.
      z <- gsub("R\\$|US\\$", "", z)
      z <- gsub("[[:space:]]", "", z)
      
      # If a comma is present, interpret Brazilian-style decimal formatting.
      if (grepl(",", z, fixed = TRUE)) {
        z <- gsub("\\.", "", z)
        z <- sub(",", ".", z, fixed = TRUE)
      }
      
      out[i] <- suppressWarnings(as.numeric(z))
    }
  }
  
  out
}

extract_year_scalar <- function(x, year_min = 1800L, year_max = 2100L) {
  if (length(x) == 0 || is.na(x)) {
    return(NA_integer_)
  }
  
  # Direct numeric interpretation first.
  direct <- suppressWarnings(as.numeric(x))
  
  if (
    length(direct) == 1 &&
    is.finite(direct) &&
    abs(direct - round(direct)) < 1e-8 &&
    direct >= year_min &&
    direct <= year_max
  ) {
    return(as.integer(round(direct)))
  }
  
  # Safe text parser. No regexpr/regmatches/substring machinery.
  s <- trimws(as.character(x))
  
  if (!nzchar(s)) {
    return(NA_integer_)
  }
  
  cleaned <- gsub("[^0-9]+", " ", s)
  tokens <- strsplit(trimws(cleaned), "[[:space:]]+")[[1]]
  
  if (length(tokens) == 0) {
    return(NA_integer_)
  }
  
  tokens <- tokens[nchar(tokens) == 4]
  
  if (length(tokens) == 0) {
    return(NA_integer_)
  }
  
  candidates <- suppressWarnings(as.integer(tokens))
  candidates <- candidates[
    !is.na(candidates) &
      candidates >= year_min &
      candidates <= year_max
  ]
  
  if (length(candidates) == 0) {
    return(NA_integer_)
  }
  
  candidates[1]
}

make_year_matrix <- function(raw, year_min, year_max) {
  nr <- nrow(raw)
  nc <- ncol(raw)
  
  yrs <- matrix(
    NA_integer_,
    nrow = nr,
    ncol = nc
  )
  
  for (i in seq_len(nr)) {
    for (j in seq_len(nc)) {
      yrs[i, j] <- extract_year_scalar(
        raw[[j]][i],
        year_min = year_min,
        year_max = year_max
      )
    }
  }
  
  yrs
}

detect_year_layout <- function(raw, year_min, year_max, minimum_hits = 5) {
  yrs <- make_year_matrix(raw, year_min, year_max)
  
  row_hits <- apply(
    yrs,
    1,
    function(z) length(unique(z[!is.na(z)]))
  )
  
  col_hits <- apply(
    yrs,
    2,
    function(z) length(unique(z[!is.na(z)]))
  )
  
  best_row <- which.max(row_hits)
  best_col <- which.max(col_hits)
  
  if (row_hits[best_row] >= minimum_hits && row_hits[best_row] >= col_hits[best_col]) {
    return(list(
      orientation = "years_across_row",
      axis_index = best_row,
      year_matrix = yrs,
      hits = row_hits[best_row]
    ))
  }
  
  if (col_hits[best_col] >= minimum_hits) {
    return(list(
      orientation = "years_down_column",
      axis_index = best_col,
      year_matrix = yrs,
      hits = col_hits[best_col]
    ))
  }
  
  stop(
    "Could not identify a reliable year axis. Best row contained ",
    row_hits[best_row], " years; best column contained ",
    col_hits[best_col], " years."
  )
}

# 3. EXACT IBGE NATIONAL-ACCOUNTS EXTRACTOR
# =============================================================================

choose_axis_series <- function(
    labels,
    exact = NULL,
    contains = NULL,
    exclude = NULL,
    series_name
) {
  labels <- normalize_text(labels)
  keep <- rep(TRUE, length(labels))
  
  if (!is.null(contains)) {
    for (p in contains) {
      keep <- keep & grepl(normalize_text(p), labels, fixed = TRUE)
    }
  }
  
  if (!is.null(exclude)) {
    for (p in exclude) {
      keep <- keep & !grepl(normalize_text(p), labels, fixed = TRUE)
    }
  }
  
  # Prefer a literal exact label when available.
  if (!is.null(exact)) {
    target <- normalize_text(exact)
    exact_idx <- which(labels == target & keep)
    if (length(exact_idx) == 1) {
      return(exact_idx)
    }
    
    # Historical sheets often attach footnote numbers to the label. In that
    # case use containment while preserving the exclusion rules.
    keep <- keep & grepl(target, labels, fixed = TRUE)
  }
  
  idx <- which(keep)
  
  if (length(idx) != 1) {
    cat("\nAvailable normalized series labels:\n")
    for (k in seq_along(labels)) {
      if (nzchar(labels[k])) {
        cat(sprintf("[%03d] %s\n", k, labels[k]))
      }
    }
    
    stop(
      "Could not uniquely identify '", series_name,
      "'. Found ", length(idx), " match(es)."
    )
  }
  
  idx
}

extract_ibge_national_accounts <- function(path) {
  sheets <- readxl::excel_sheets(path)
  preferred <- "I.2.1"
  
  if (!(preferred %in% sheets)) {
    stop(
      "Expected sheet 'I.2.1' in ", basename(path),
      ". Available sheets include:\n",
      paste(sheets, collapse = ", ")
    )
  }
  
  raw <- read_raw_sheet(path, preferred)
  
  layout <- detect_year_layout(
    raw,
    year_min = 1947,
    year_max = 1989,
    minimum_hits = 10
  )
  
  cat(
    "\nNational-accounts sheet: ", preferred,
    "\nDetected layout: ", layout$orientation,
    "\n",
    sep = ""
  )
  
  wanted_years <- 1967:1972
  
  # ---------------------------------------------------------------------------
  # CASE A: years run across columns; economic series run down rows.
  # ---------------------------------------------------------------------------
  if (layout$orientation == "years_across_row") {
    year_row <- layout$axis_index
    year_vector <- layout$year_matrix[year_row, ]
    year_cols <- match(wanted_years, year_vector)
    
    if (any(is.na(year_cols))) {
      stop(
        "Could not find all years 1967-1972 in I.2.1. Found: ",
        paste(sort(unique(year_vector[!is.na(year_vector)])), collapse = ", ")
      )
    }
    
    first_year_col <- min(which(!is.na(year_vector)))
    label_columns <- if (first_year_col > 1) seq_len(first_year_col - 1) else 1
    candidate_rows <- seq.int(year_row + 1, nrow(raw))
    
    row_labels <- vapply(
      candidate_rows,
      function(i) {
        normalize_text(
          paste(
            unlist(raw[i, label_columns, drop = FALSE], use.names = FALSE),
            collapse = " "
          )
        )
      },
      character(1)
    )
    
    gdp_rel <- choose_axis_series(
      row_labels,
      exact = "Produto Interno Bruto",
      contains = "Produto Interno Bruto",
      exclude = c(
        "per capita",
        "custo de fatores",
        "precos constantes",
        "produto real",
        "produto nacional"
      ),
      series_name = "Produto Interno Bruto"
    )
    
    exp_rel <- choose_axis_series(
      row_labels,
      exact = "Exportacao de bens e servicos",
      contains = "Exportacao de bens e servicos",
      series_name = "Exportacao de bens e servicos"
    )
    
    imp_rel <- choose_axis_series(
      row_labels,
      exact = "Menos: Importacao de bens e servicos",
      contains = c("Menos", "Importacao de bens e servicos"),
      series_name = "Menos: Importacao de bens e servicos"
    )
    
    gdp_row <- candidate_rows[gdp_rel]
    exp_row <- candidate_rows[exp_rel]
    imp_row <- candidate_rows[imp_rel]
    
    cat(
      "Matched GDP row:     ", gdp_row, " -> ", row_labels[gdp_rel], "\n",
      "Matched export row:  ", exp_row, " -> ", row_labels[exp_rel], "\n",
      "Matched import row:  ", imp_row, " -> ", row_labels[imp_rel], "\n",
      sep = ""
    )
    
    value_at <- function(row_index) {
      vapply(
        year_cols,
        function(j) as_number_safe(raw[[j]][row_index]),
        numeric(1)
      )
    }
    
    out <- data.frame(
      year = wanted_years,
      gdp_current = value_at(gdp_row),
      exports_current = value_at(exp_row),
      imports_current_raw = value_at(imp_row),
      stringsAsFactors = FALSE
    )
  }
  
  # ---------------------------------------------------------------------------
  # CASE B: years run down rows; economic series run across columns.
  # ---------------------------------------------------------------------------
  if (layout$orientation == "years_down_column") {
    year_col <- layout$axis_index
    year_vector <- layout$year_matrix[, year_col]
    year_rows <- match(wanted_years, year_vector)
    
    if (any(is.na(year_rows))) {
      stop(
        "Could not find all years 1967-1972 in I.2.1. Found: ",
        paste(sort(unique(year_vector[!is.na(year_vector)])), collapse = ", ")
      )
    }
    
    first_year_row <- min(which(!is.na(year_vector)))
    header_rows <- if (first_year_row > 1) seq_len(first_year_row - 1) else 1
    
    col_labels <- vapply(
      seq_len(ncol(raw)),
      function(j) {
        normalize_text(
          paste(
            unlist(raw[header_rows, j, drop = FALSE], use.names = FALSE),
            collapse = " "
          )
        )
      },
      character(1)
    )
    
    gdp_col <- choose_axis_series(
      col_labels,
      exact = "Produto Interno Bruto",
      contains = "Produto Interno Bruto",
      exclude = c(
        "per capita",
        "custo de fatores",
        "precos constantes",
        "produto real",
        "produto nacional"
      ),
      series_name = "Produto Interno Bruto"
    )
    
    exp_col <- choose_axis_series(
      col_labels,
      exact = "Exportacao de bens e servicos",
      contains = "Exportacao de bens e servicos",
      series_name = "Exportacao de bens e servicos"
    )
    
    imp_col <- choose_axis_series(
      col_labels,
      exact = "Menos: Importacao de bens e servicos",
      contains = c("Menos", "Importacao de bens e servicos"),
      series_name = "Menos: Importacao de bens e servicos"
    )
    
    cat(
      "Matched GDP column:     ", gdp_col, " -> ", col_labels[gdp_col], "\n",
      "Matched export column:  ", exp_col, " -> ", col_labels[exp_col], "\n",
      "Matched import column:  ", imp_col, " -> ", col_labels[imp_col], "\n",
      sep = ""
    )
    
    out <- data.frame(
      year = wanted_years,
      gdp_current = as_number_safe(raw[[gdp_col]][year_rows]),
      exports_current = as_number_safe(raw[[exp_col]][year_rows]),
      imports_current_raw = as_number_safe(raw[[imp_col]][year_rows]),
      stringsAsFactors = FALSE
    )
  }
  
  out$imports_current <- abs(out$imports_current_raw)
  
  if (any(!is.finite(out$gdp_current)) || any(out$gdp_current <= 0)) {
    stop("GDP extraction from I.2.1 produced missing/non-positive values.")
  }
  
  if (any(!is.finite(out$exports_current))) {
    stop("Export extraction from I.2.1 produced missing values.")
  }
  
  if (any(!is.finite(out$imports_current))) {
    stop("Import extraction from I.2.1 produced missing values.")
  }
  
  out$x_y <- out$exports_current / out$gdp_current
  out$m_y <- out$imports_current / out$gdp_current
  
  attr(out, "sheet") <- preferred
  attr(out, "layout") <- layout$orientation
  out
}

# =============================================================================
# 4. OPTIONAL EXTERNAL-SECTOR FOB EXTRACTOR
# =============================================================================
# This part is diagnostic only. If parsing fails, Script 06 continues.
# Exact labels verified from the workbook:
#   "Exportacoes (FOB)"
#   "Importacoes (FOB)"
#   "Saldo Comercial"

build_axis_labels <- function(raw, layout) {
  if (layout$orientation == "years_down_column") {
    year_col <- layout$axis_index
    year_values <- layout$year_matrix[, year_col]
    year_rows <- which(!is.na(year_values))
    first_year_row <- min(year_rows)
    
    header_rows <- if (first_year_row > 1) seq_len(first_year_row - 1) else 1
    
    labels <- vapply(
      seq_len(ncol(raw)),
      function(j) {
        paste(
          normalize_text(
            unlist(raw[header_rows, j, drop = FALSE], use.names = FALSE)
          ),
          collapse = " "
        ) |> normalize_text()
      },
      character(1)
    )
    
    return(list(
      labels = labels,
      year_values = year_values,
      year_rows = year_rows
    ))
  }
  
  year_row <- layout$axis_index
  year_values <- layout$year_matrix[year_row, ]
  year_cols <- which(!is.na(year_values))
  first_year_col <- min(year_cols)
  
  label_cols <- if (first_year_col > 1) seq_len(first_year_col - 1) else 1
  
  labels <- vapply(
    seq_len(nrow(raw)),
    function(i) {
      paste(
        normalize_text(
          unlist(raw[i, label_cols, drop = FALSE], use.names = FALSE)
        ),
        collapse = " "
      ) |> normalize_text()
    },
    character(1)
  )
  
  list(
    labels = labels,
    year_values = year_values,
    year_cols = year_cols
  )
}

choose_label_index <- function(labels, exact_phrase, label_name) {
  target <- normalize_text(exact_phrase)
  
  exact <- which(labels == target)
  if (length(exact) == 1) return(exact)
  
  contains <- which(grepl(target, labels, fixed = TRUE))
  if (length(contains) == 1) return(contains)
  
  stop(
    "Could not uniquely identify '", label_name,
    "'. Matches found: ", length(contains), "."
  )
}

extract_external_fob_optional <- function(path) {
  tryCatch({
    sheets <- readxl::excel_sheets(path)
    
    # The workbook itself identifies this historical table as sheet 2.
    sheet <- if ("2" %in% sheets) "2" else sheets[2]
    
    raw <- read_raw_sheet(path, sheet)
    
    layout <- detect_year_layout(
      raw,
      year_min = 1901,
      year_max = 2000,
      minimum_hits = 20
    )
    
    cat(
      "\nExternal-sector sheet: ", sheet,
      "\nDetected layout: ", layout$orientation,
      "\n",
      sep = ""
    )
    
    info <- build_axis_labels(raw, layout)
    
    exp_idx <- choose_label_index(
      info$labels,
      "Exportacoes FOB",
      "Exportacoes (FOB)"
    )
    
    imp_idx <- choose_label_index(
      info$labels,
      "Importacoes FOB",
      "Importacoes (FOB)"
    )
    
    if (layout$orientation == "years_down_column") {
      year_col <- layout$axis_index
      years <- layout$year_matrix[, year_col]
      rows <- which(!is.na(years) & years >= 1967 & years <= 1972)
      
      out <- data.frame(
        year = years[rows],
        exports_fob_usd_million = as_number_safe(raw[[exp_idx]][rows]),
        imports_fob_usd_million = abs(as_number_safe(raw[[imp_idx]][rows])),
        stringsAsFactors = FALSE
      )
    } else {
      year_row <- layout$axis_index
      years <- layout$year_matrix[year_row, ]
      cols <- which(!is.na(years) & years >= 1967 & years <= 1972)
      
      out <- data.frame(
        year = years[cols],
        exports_fob_usd_million = vapply(
          cols,
          function(j) as_number_safe(raw[[j]][exp_idx]),
          numeric(1)
        ),
        imports_fob_usd_million = abs(vapply(
          cols,
          function(j) as_number_safe(raw[[j]][imp_idx]),
          numeric(1)
        )),
        stringsAsFactors = FALSE
      )
    }
    
    out <- out[order(out$year), ]
    out$trade_balance_usd_million <-
      out$exports_fob_usd_million - out$imports_fob_usd_million
    
    out
  }, error = function(e) {
    warning(
      paste0(
        "Optional FOB diagnostic could not be read and will be skipped. ",
        "This does NOT affect x_y, cf_y or z_y. Details: ",
        conditionMessage(e)
      )
    )
    NULL
  })
}

summary_stats <- function(x) {
  x <- x[is.finite(x)]
  
  c(
    n = length(x),
    mean = mean(x),
    median = median(x),
    sd = if (length(x) > 1) sd(x) else NA_real_,
    min = min(x),
    max = max(x)
  )
}

make_parameter_row <- function(parameter, values, interpretation, method) {
  s <- summary_stats(values)
  
  data.frame(
    parameter = parameter,
    value = unname(s["mean"]),
    n = as.integer(s["n"]),
    median = unname(s["median"]),
    annual_sd = unname(s["sd"]),
    min = unname(s["min"]),
    max = unname(s["max"]),
    sample = "1968-1972",
    method = method,
    interpretation = interpretation,
    recommended_baseline = "YES",
    stringsAsFactors = FALSE
  )
}

# =============================================================================
# 07_alpha_mu_lambda_s.R
# =============================================================================
#
# PURPOSE
# -------
# Calibrate / derive:
#
#   alpha     = imported-consumption weight in the household CES aggregator
#   mu        = imported-intermediate-input cost share in production
#   lambda_s  = total terms-of-trade loading in marginal cost
#
# MODEL MAPPING
# -------------
#
#   alpha_t = cf_y_t / c_y_t
#
# where:
#   cf_y_t = imported consumer goods / GDP   (from Script 06)
#   c_y_t  = private household consumption / GDP
#
# and:
#
#   mu_t = z_y_t / (labor_share_t + z_y_t)
#
# where:
#   z_y_t = imported intermediate goods / GDP   (from Script 06)
#
# Finally:
#
#   lambda_s = mu + (1-mu)*alpha
#
# BASELINE SAMPLE
# ---------------
# 1968-1972
#
# IMPORTANT FIX IN THIS VERSION
# -----------------------------
# Script 07 no longer tries to identify GDP again inside the IBGE workbook.
# Script 06 already extracted and validated GDP, so this script reuses
# gdp_current from:
#
#   data/processed/trade_steady_state_annual_1967_1972.csv
#
# The IBGE workbook is opened only to identify household/private consumption.
# This avoids the previous ambiguity from multiple rows containing the phrase
# "Produto Interno Bruto".
# =============================================================================

norm_text <- function(x) {
  
  x <- as.character(x)
  x[is.na(x)] <- ""
  
  y <- suppressWarnings(
    iconv(
      x,
      from = "",
      to = "ASCII//TRANSLIT"
    )
  )
  
  y[is.na(y)] <- x[is.na(y)]
  
  y <- tolower(y)
  y <- gsub("[\r\n\t]+", " ", y)
  y <- gsub("[^a-z0-9]+", " ", y)
  y <- trimws(
    gsub(
      "[[:space:]]+",
      " ",
      y
    )
  )
  
  y
}


as_num <- function(x) {
  
  if (is.numeric(x)) {
    return(as.numeric(x))
  }
  
  z <- trimws(as.character(x))
  
  z[
    z %in% c(
      "",
      "NA",
      "N/A",
      "-",
      "--"
    )
  ] <- NA_character_
  
  ans <- suppressWarnings(
    as.numeric(z)
  )
  
  miss <- is.na(ans) & !is.na(z)
  
  if (any(miss)) {
    
    z2 <- z[miss]
    
    z2 <- gsub(
      "R\\$|US\\$",
      "",
      z2
    )
    
    z2 <- gsub(
      "[[:space:]]",
      "",
      z2
    )
    
    has_comma <- grepl(
      ",",
      z2,
      fixed = TRUE
    )
    
    z2[has_comma] <- gsub(
      "\\.",
      "",
      z2[has_comma]
    )
    
    z2[has_comma] <- sub(
      ",",
      ".",
      z2[has_comma],
      fixed = TRUE
    )
    
    ans[miss] <- suppressWarnings(
      as.numeric(z2)
    )
  }
  
  ans
}


as_year_flexible <- function(
    x,
    year_min = 1800,
    year_max = 2100
) {
  
  s <- as.character(x)
  s[is.na(x)] <- NA_character_
  s <- trimws(s)
  
  out <- rep(
    NA_integer_,
    length(s)
  )
  
  direct <- suppressWarnings(
    as.numeric(s)
  )
  
  good_direct <- (
    !is.na(direct) &
      abs(direct - round(direct)) < 1e-8 &
      direct >= year_min &
      direct <= year_max
  )
  
  out[good_direct] <- as.integer(
    round(
      direct[good_direct]
    )
  )
  
  left <- which(
    is.na(out) &
      !is.na(s) &
      nzchar(s)
  )
  
  if (length(left)) {
    
    for (ii in left) {
      
      cleaned <- gsub(
        "[^0-9]+",
        " ",
        s[ii]
      )
      
      tokens <- strsplit(
        trimws(cleaned),
        "[[:space:]]+"
      )[[1]]
      
      tokens <- tokens[
        nzchar(tokens) &
          nchar(tokens) == 4
      ]
      
      if (!length(tokens)) {
        next
      }
      
      candidates <- suppressWarnings(
        as.integer(tokens)
      )
      
      candidates <- candidates[
        !is.na(candidates) &
          candidates >= year_min &
          candidates <= year_max
      ]
      
      if (length(candidates)) {
        out[ii] <- candidates[1]
      }
    }
  }
  
  out
}


stats_row <- function(x) {
  
  x <- x[
    is.finite(x)
  ]
  
  data.frame(
    n = length(x),
    mean = mean(x),
    median = median(x),
    sd = if (length(x) > 1) sd(x) else NA_real_,
    min = min(x),
    max = max(x)
  )
}

extract_values_for_row <- function(
    row_id,
    years_needed
) {
  
  out <- rep(
    NA_real_,
    length(years_needed)
  )
  
  for (k in seq_along(years_needed)) {
    
    y <- years_needed[k]
    
    cols <- which(
      year_vector == y
    )
    
    if (length(cols) != 1) {
      
      stop(
        "Expected exactly one column for year ",
        y,
        " in I.2.1."
      )
    }
    
    out[k] <- as_num(
      na_raw[
        row_id,
        cols
      ][[1]]
    )
  }
  
  out
}

  candidate_diagnostics$mean_c_y[j] <-

        c_y_candidate,

    }

    if (

        is.finite(
          c_y_candidate
        )
      )

        c_y_candidate,
        na.rm = TRUE

# ---------------------------------------------------------------------------
# If several plausible rows remain, use a deterministic priority ranking
# based on the economic label.
# ---------------------------------------------------------------------------

label_priority <- function(label) {
  
  if (
    grepl(
      "^consumo final das familias$",
      label
    )
  ) {
    return(1)
  }
  
  if (
    grepl(
      "consumo final das familias",
      label
    )
  ) {
    return(2)
  }
  
  if (
    grepl(
      "^consumo pessoal$",
      label
    )
  ) {
    return(3)
  }
  
  if (
    grepl(
      "consumo pessoal",
      label
    )
  ) {
    return(4)
  }
  
  if (
    grepl(
      "despesa de consumo das familias",
      label
    )
  ) {
    return(5)
  }
  
  if (
    grepl(
      "consumo das familias",
      label
    )
  ) {
    return(6)
  }
  
  99
}


candidate_diagnostics$priority <- vapply(
  candidate_diagnostics$label,
  label_priority,
  numeric(1)
)

  annual$c_y

    annual$alpha >= 1

  )

# =============================================================================
# 8. CALIBRATE mu
# =============================================================================
#
# Production:
#
#   Y = A*N^(1-mu)*Z^mu
#
# Cost minimization:
#
#   mu = P_Z Z / (W N + P_Z Z)
#
# Because both z_y and labor share use GDP as denominator:
#
#   mu = z_y / (labor_share + z_y)

annual$mu_adjusted <- with(
  annual,
  z_y /
    (
      labor_share_adjusted +
        z_y
    )
)


annual$mu_employee_only <- with(
  annual,
  z_y /
    (
      employee_comp_share_pc +
        z_y
    )
)

  any(
    annual$mu_adjusted <= 0 |
    annual$mu_adjusted >= 1
  )
) {
  
  stop(
    "Calculated baseline mu is outside (0,1)."
  )
}

# =============================================================================
# 9. DERIVE lambda_s
# =============================================================================

annual$lambda_s_annual <- with(
  annual,
  mu_adjusted +
    (
      1 -
        mu_adjusted
    ) *
    alpha
)

# -------
# Build a reproducible historical database for the Brazil external-debt block

# ---------

# series. Different papers use different debt concepts (registered debt,
# gross debt, net debt, broader total debt including short-term items, etc.).
#
# The next script (08) will decide which empirical concept maps best to:
#

# -----------------------------------------------------------------------------
# Safe parser for BCB values
# -----------------------------------------------------------------------------
#
# Handles:
#   12571.5   -> 12571.5
#   12571,5   -> 12571.5
#   12.571,5  -> 12571.5
#
# Critically, decimal points are NOT removed unless a comma is also present,
# so values like 12571.5 do not become 125715.

parse_bcb_value <- function(x) {
  
  x <- trimws(
    as.character(x)
  )
  
  x[
    x %in% c(
      "",
      "NA",
      "null"
    )
  ] <- NA_character_
  
  out <- rep(
    NA_real_,
    length(x)
  )
  
  for (i in seq_along(x)) {
    
    z <- x[i]
    
    if (is.na(z)) {
      next
    }
    
    has_comma <- grepl(
      ",",
      z,
      fixed = TRUE
    )
    
    has_dot <- grepl(
      ".",
      z,
      fixed = TRUE
    )
    
    # Brazilian thousands + decimal:
    # 12.571,5 -> 12571.5
    if (
      has_comma &&
      has_dot
    ) {
      
      z <- gsub(
        ".",
        "",
        z,
        fixed = TRUE
      )
      
      z <- sub(
        ",",
        ".",
        z,
        fixed = TRUE
      )
      
      # Decimal comma only:
      # 12571,5 -> 12571.5
    } else if (
      has_comma
    ) {
      
      z <- sub(
        ",",
        ".",
        z,
        fixed = TRUE
      )
      
      # Decimal point only:
      # 12571.5
      # Leave unchanged.
    }
    
    out[i] <- suppressWarnings(
      as.numeric(z)
    )
  }
  
  out
}

# Validate and filter BCB data
# -----------------------------------------------------------------------------

if (
  is.null(
    bcb_all
  ) ||
  !nrow(
    bcb_all
  )
) {
  
  warning(
    paste(

      "Historical-paper tables remain usable, but the official BCB",

  bcb_3560 <- data.frame(

        character(0)
      ),
    
    year =
      integer(0),
    
    registered_external_debt_usd_million =
      numeric(0),
    
    stringsAsFactors = FALSE
  )
  
} else {
  
  bcb_all$year <- as.integer(
    format(
      bcb_all$date,
      "%Y"
    )
  )
  
  bcb_3560 <- bcb_all[
    bcb_all$year >= 1964 &
      bcb_all$year <= 1980,
    ,
    drop = FALSE
  ]
  
  names(
    bcb_3560
  )[
    names(
      bcb_3560
    ) ==
      "value"
  ] <-
    "registered_external_debt_usd_million"
  
  bcb_3560$source_id <-
    "BCB_SGS_3560"
  
  bcb_3560 <- bcb_3560[
    ,
    c(
      "source_id",
      "date",
      "year",
      "registered_external_debt_usd_million"
    )
  ]
  
  bcb_3560 <- bcb_3560[
    order(
      bcb_3560$year
    ),
    ,
    drop = FALSE
  ]
  
  expected_years <-
    1964:1980
  
  missing_years <- setdiff(
    expected_years,
    bcb_3560$year
  )
  
  duplicated_years <- unique(
    bcb_3560$year[
      duplicated(
        bcb_3560$year
      )
    ]
  )
  
  if (
    length(
      missing_years
    )
  ) {
    
    warning(
      "BCB SGS 3560 is missing expected year(s): ",
      paste(
        missing_years,
        collapse = ", "
      )
    )
  }
  
  if (
    length(
      duplicated_years
    )
  ) {
    
    warning(
      "BCB SGS 3560 contains duplicated year(s): ",
      paste(
        duplicated_years,
        collapse = ", "
      )
    )
  }
  
  if (
    any(
      !is.finite(
        bcb_3560$registered_external_debt_usd_million
      )
    )
  ) {
    
    warning(
      "BCB SGS 3560 contains non-numeric or missing debt observations."
    )
  }

  # Deterministic local validation.
  #
  # These checks are intentionally simple and transparent rather than
  # cryptographic: they detect the historical decimal-parsing failures that
  # previously turned values such as 6621.3 into 66213.
  expected_bcb_years <- 1964:1980
  expected_bcb_sum <- 299560.1
  expected_bcb_weighted_sum <- 3977069.8

  bcb_checksum_sum <- sum(
    bcb_3560$registered_external_debt_usd_million,
    na.rm = TRUE
  )

  bcb_checksum_weighted <- sum(
    seq_len(nrow(bcb_3560)) *
      bcb_3560$registered_external_debt_usd_million,
    na.rm = TRUE
  )

  if (
    identical(bcb_3560$year, expected_bcb_years) &&
    (
      abs(bcb_checksum_sum - expected_bcb_sum) > 1e-6 ||
      abs(bcb_checksum_weighted - expected_bcb_weighted_sum) > 1e-6
    )
  ) {
    stop(
      paste(
        "BCB SGS 3560 checksum validation failed.",
        "Expected sum=", expected_bcb_sum,
        "and weighted sum=", expected_bcb_weighted_sum,
        "but obtained", bcb_checksum_sum,
        "and", bcb_checksum_weighted, ".",
        "Check decimal parsing or the cached/downloaded source."
      )
    )
  }

  # Historical cross-check against independent literature.
  historical_check <- data.frame(
    
    year = c(
      1968,
      1972,
      1973,
      1979,
      1980
    ),
    
    literature_value_usd_million = c(
      3780,
      9521,
      12571,
      49904,
      53847
    )
  )
  
  historical_check$bcb_value_usd_million <-
    bcb_3560$registered_external_debt_usd_million[
      match(
        historical_check$year,
        bcb_3560$year
      )
    ]
  
  historical_check$difference_usd_million <-
    historical_check$bcb_value_usd_million -
    historical_check$literature_value_usd_million
  
  historical_check$pct_difference <-
    100 *
    historical_check$difference_usd_million /
    historical_check$literature_value_usd_million
  
  write.csv(
    historical_check,
    file.path(
      processed_dir,
      "bcb3560_historical_source_check.csv"
    ),
    row.names = FALSE,
    na = ""
  )
  
  cat(
    "\n============================================================\n"
  )
  
  cat(
    "BCB SGS 3560 DOWNLOAD SUCCESSFUL\n"
  )
  
  cat(
    "============================================================\n"
  )
  
  cat(
    "\nSeries: Dívida externa registrada - Total - anual\n"
  )
  
  cat(
    "Unit: US$ million\n"
  )
  
  cat(
    "Window retained: 1964-1980\n"
  )
  
  cat(
    "Observations retained: ",
    nrow(
      bcb_3560
    ),
    "\n\n",
    sep = ""
  )
  
  print(
    bcb_3560,
    row.names = FALSE
  )
  
  cat(
    "\nHistorical literature cross-check:\n\n"
  )
  
  print(
    historical_check,
    row.names = FALSE
  )
}

  row.names = FALSE,
  na = ""

# 10. HELPER FOR YEAR MATCHING
# =============================================================================

get_by_year <- function(df, value_col, years) {
  
  idx <- match(
    years,
    df$year
  )
  
  out <- rep(
    NA_real_,
    length(years)
  )
  
  good <- !is.na(idx)
  
  out[good] <- df[
    idx[good],
    value_col
  ]
  
  out
}

# the script extracts the series automatically. Otherwise it writes the
# verified 1968-1980 values as a transparent fallback transcription.
#

aiolfi_local_file <- file.path(
  raw_debt_dir,
  "LAC4_database_Aiolfi-Catao-Timmermann_1870-2004.csv"
)


extract_aiolfi_ext_spread <- function(path) {
  
  raw_aiolfi <- read.csv(
    path,
    header = FALSE,
    stringsAsFactors = FALSE,
    check.names = FALSE
  )
  
  year_header_row <- which(
    apply(
      raw_aiolfi,
      1,
      function(z) any(trimws(as.character(z)) == "Year", na.rm = TRUE)
    )
  )
  
  if (!length(year_header_row)) {
    stop("Could not identify the Aiolfi header row containing 'Year'.")
  }
  
  h <- year_header_row[1]
  header_values <- trimws(as.character(raw_aiolfi[h, ]))
  
  year_col <- which(header_values == "Year")[1]
  spread_col <- which(header_values == "Ext.Spread")[1]
  
  if (is.na(year_col) || is.na(spread_col)) {
    stop("Could not identify Year and Ext.Spread columns in the Aiolfi CSV.")
  }
  
  dat <- raw_aiolfi[(h + 2):nrow(raw_aiolfi), c(year_col, spread_col)]
  names(dat) <- c("year", "effective_external_cost_spread_pp")
  
  dat$year <- suppressWarnings(as.integer(trimws(as.character(dat$year))))
  dat$effective_external_cost_spread_pp <- suppressWarnings(as.numeric(trimws(as.character(dat$effective_external_cost_spread_pp))))
  
  dat <- dat[
    is.finite(dat$year) &
      dat$year >= 1968 &
      dat$year <= 1986,
    ,
    drop = FALSE
  ]
  
  dat$source_id <- "AIOLFI_CATAO_TIMMERMANN_2010"
  dat$concept <- "published_effective_cost_spread_not_marginal_loan_spread"
  
  dat
}


if (file.exists(aiolfi_local_file)) {
  
  aiolfi_ext_spread <- tryCatch(
    extract_aiolfi_ext_spread(aiolfi_local_file),
    error = function(e) {
      warning(
        "Aiolfi CSV found but automatic extraction failed: ",
        conditionMessage(e),
        ". Using verified fallback values for 1968-1980."
      )
      NULL
    }
  )
  
} else {
  
  aiolfi_ext_spread <- NULL
}

# =============================================================================
# 08_external_debt_and_financing_wedge_parameters.R
# =============================================================================
#
# PURPOSE
# -------
# Finalize the parsimonious external-debt / external-financing block for
# Paper I while preserving the same structural architecture for Papers II-III.
#
# STRUCTURAL BASELINE
# -------------------
#
#   rp_t = phiD * d_t + eps_rp,t
#
# phiDD is not part of the live model specification.
#
# Parameter strategy:
#
#   dbar  -> Brazil-specific historical calibration
#   phiD  -> small positive quarterly SOE literature calibration
#   rp_t  -> historically disciplined EXTERNAL FINANCING WEDGE
#
#
# PAPER-I OBSERVABLE
# ------------------
#
# Brazil's syndicated loans were priced approximately as:
#
#   Brazil marginal external rate
#     = LIBOR + Brazil contractual spread over LIBOR
#
# The model's world rate istar is already mapped to TB3MS.
#
# Therefore the model-consistent observed financing wedge is:
#
#   rp_obs
#     = (LIBOR - TB3MS)
#       + Brazil spread over LIBOR
#
# so that:
#
#   TB3MS + rp_obs
#     = LIBOR + Brazil spread.
#
#
# IMPORTANT
# ---------
#
# This script DOES NOT estimate phiD from the 1970s Brazilian data.
#
# Why?
#
#   1) the usable marginal-spread sample is short;
#   2) debt and spreads are jointly determined;
#   3) global credit supply strongly affected Brazil's borrowing conditions;
#   4) the economic role of phiD in this model is primarily stationarity and
#      long-run debt discipline;
#   5) historically important movements in financing conditions are allowed to
#      enter through eps_rp,t.
#
#
# BASELINE phiD
# -------------
#
# We use:
#
#   phiD = 0.001
#
# at quarterly frequency.
#
# This is a small debt-elastic risk-premium calibration used directly in
# quarterly small-open-economy work and described as comparable to the
# Schmitt-Grohe & Uribe (2003) debt-elastic-premium calibration.
#
# Schmitt-Grohe & Uribe use:
#
#   p(d) = psi2 * ( exp(d-dbar) - 1 )
#
# whose first-order approximation around dbar is:
#
#   p(d) ~= psi2 * (d-dbar).
#
# Thus our linear rule has the same local form.
#
# We intentionally use phiD as a small structural/stationarity parameter, not
# as an attempt to explain the level of historical Brazilian sovereign spreads.

effective_quarterly <- function(r_annual_decimal) {
  
  out <- rep(NA_real_, length(r_annual_decimal))
  
  ok <- is.finite(r_annual_decimal) &
    (1 + r_annual_decimal > 0)
  
  out[ok] <- (1 + r_annual_decimal[ok])^(1 / 4) - 1
  
  out
}


get_by_year <- function(df, value_col, years) {
  
  idx <- match(years, df$year)
  
  out <- rep(NA_real_, length(years))
  
  good <- !is.na(idx)
  
  out[good] <- df[idx[good], value_col]
  
  out
}

# =============================================================================
# 3. CALIBRATE dbar
# =============================================================================
#
# Baseline:
#
#   mean net external debt / GDP, 1970-1972
#
# This is a pre-first-oil-shock steady-state calibration.
#
# Gross debt/GDP is retained as robustness.
#

dbar_sample <- debt[
  debt$year >= 1970 &
    debt$year <= 1972,
  ,
  drop = FALSE
]

  stop("Expected three complete net debt/GDP observations for 1970-1972.")
}


dbar <- mean(
  dbar_sample$net_debt_ratio
)


dbar_net_median <- median(
  dbar_sample$net_debt_ratio
)


dbar_net_1972 <- dbar_sample$net_debt_ratio[
  dbar_sample$year == 1972

  dbar_sample$gross_debt_ratio
)


dbar_gross_median <- median(
  dbar_sample$gross_debt_ratio
)


dbar_gross_1972 <- dbar_sample$gross_debt_ratio[
  dbar_sample$year == 1972
]

# =============================================================================
# 7. PAPER-I BASELINE MARGINAL FINANCING WEDGE
# =============================================================================
#
# Historical marginal Brazil rate:
#
#   r_BR = LIBOR + Brazil spread
#
# Model-consistent wedge over TB3MS:
#
#   rp_obs = r_BR - TB3MS
#
# At quarterly frequency we transform BOTH gross annual rates separately:
#
#   r_BR,q  = (1+r_BR,a)^(1/4)-1
#   i*,q    = (1+i*,a)^(1/4)-1
#
#   rp_obs,q = r_BR,q - i*,q
#

paper1_rp <- merge(
  bacha,
  tb3ms_annual,
  by = "year",
  all.x = TRUE,
  sort = TRUE
)


paper1_rp$brazil_marginal_rate_annual_decimal <-
  (
    paper1_rp$libor_annual_pct +
      paper1_rp$brazil_spread_over_libor_pp
  ) / 100


paper1_rp$libor_annual_decimal <-
  paper1_rp$libor_annual_pct / 100


paper1_rp$brazil_spread_over_libor_decimal <-
  paper1_rp$brazil_spread_over_libor_pp / 100


paper1_rp$global_bank_wedge_over_tb3ms_annual_pp <-
  paper1_rp$libor_annual_pct -
  paper1_rp$tb3ms_annual_pct


paper1_rp$rp_model_annual_pp <-
  paper1_rp$libor_annual_pct +
  paper1_rp$brazil_spread_over_libor_pp -
  paper1_rp$tb3ms_annual_pct


paper1_rp$brazil_marginal_rate_quarterly_effective <-
  effective_quarterly(
    paper1_rp$brazil_marginal_rate_annual_decimal
  )


paper1_rp$rp_model_quarterly <-
  paper1_rp$brazil_marginal_rate_quarterly_effective -
  paper1_rp$tb3ms_quarterly_effective


paper1_rp$rp_model_quarterly_bp <-
  10000 *
  paper1_rp$rp_model_quarterly


paper1_rp$d_gap_net <-
  get_by_year(
    debt,
    "d_gap_net",
    paper1_rp$year
  )


paper1_rp$structural_debt_premium_quarterly <-
  phiD *
  paper1_rp$d_gap_net


paper1_rp$structural_debt_premium_quarterly_bp <-
  10000 *
  paper1_rp$structural_debt_premium_quarterly


# =============================================================================
# 8. 1972 PRE-SHOCK CENTERING ANCHOR
# =============================================================================
#
# The 1972 spread observation is approximate / mid-year.
#
# We use it only to define a transparent pre-shock normalization:
#
#   rp_gap_t = rp_obs_t - rp_anchor_1972
#
# It is NOT treated as a high-precision annual estimate.
#

anchor$tb3ms_annual_pct <-
  get_by_year(
    tb3ms_annual,
    "tb3ms_annual_pct",
    anchor$year
  )


anchor$tb3ms_annual_decimal <-
  anchor$tb3ms_annual_pct / 100


anchor$brazil_marginal_rate_annual_decimal <-
  (
    anchor$libor_annual_pct +
      anchor$brazil_spread_over_libor_pp
  ) / 100


anchor$brazil_marginal_rate_quarterly_effective <-
  effective_quarterly(
    anchor$brazil_marginal_rate_annual_decimal
  )


anchor$tb3ms_quarterly_effective <-
  effective_quarterly(
    anchor$tb3ms_annual_decimal
  )


anchor$rp_anchor_quarterly <-
  anchor$brazil_marginal_rate_quarterly_effective -
  anchor$tb3ms_quarterly_effective


anchor$rp_anchor_quarterly_bp <-
  10000 *
  anchor$rp_anchor_quarterly


rp_anchor_q <- anchor$rp_anchor_quarterly[1]


# =============================================================================
# 9. IMPLIED PAPER-I HISTORICAL eps_rp
# =============================================================================
#
# The model equation is expressed around steady state:
#
#   rp_gap_t = phiD * d_gap_t + eps_rp,t
#
# Therefore:
#
#   eps_rp,t = rp_gap_t - phiD*d_gap_t
#
# These are ANNUAL historical observations expressed in quarterly rate units.
#
# DO NOT repeat each value four times and pretend that creates quarterly data.
#

paper1_rp$rp_gap_from_1972_anchor_quarterly <-
  paper1_rp$rp_model_quarterly -
  rp_anchor_q


paper1_rp$rp_gap_from_1972_anchor_quarterly_bp <-
  10000 *
  paper1_rp$rp_gap_from_1972_anchor_quarterly


paper1_rp$eps_rp_implied_quarterly_rate_units <-
  paper1_rp$rp_gap_from_1972_anchor_quarterly -
  paper1_rp$structural_debt_premium_quarterly


paper1_rp$eps_rp_implied_bp <-
  10000 *
  paper1_rp$eps_rp_implied_quarterly_rate_units


# =============================================================================

  (

  (
    rp_robust$libor_annual_pct +
      rp_robust$sachs_brazil_spread_over_libor_pp
  ) / 100


rp_robust$bacha_rp_q <-
  effective_quarterly(
    rp_robust$bacha_brazil_rate_annual
  ) -
  rp_robust$tb3ms_quarterly_effective


rp_robust$sachs_rp_q <-
  effective_quarterly(
    rp_robust$sachs_brazil_rate_annual
  ) -
  rp_robust$tb3ms_quarterly_effective


rp_robust$bacha_rp_bp <-
  10000 *
  rp_robust$bacha_rp_q


rp_robust$sachs_rp_bp <-
  10000 *
  rp_robust$sachs_rp_q
